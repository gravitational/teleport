```hcl
resource "teleport_provision_token" "gcp-token" {
  version = "v2"
  metadata = {
    name = "gcp-token"
  }

  labels = {
    // This label is added on the Teleport side by default
    "teleport.dev/origin" = "dynamic" 
  }

  spec = {
    // use the minimal set of roles required (e.g. Node, Proxy, App, Kube, DB, WindowsDesktop)
    roles       = ["Node"]
    join_method = "gcp"
    gcp = {
      allow = [
        {
          // The GCP project ID(s) allowed to join.
          project_ids      = ["example-project-id"]
          // (Optional) The locations that VMs can join from. Note: both regions and
          // zones are accepted. Only supported for GCP VMs.
          locations        = ["us-west1", "us-west2-a"]
          // (Optional) The email addresses of service accounts allowed to join.
          service_accounts = ["example-service-account@example-project-id.iam.gserviceaccount.com"]
        },
      ]
    }
  }
}
```
